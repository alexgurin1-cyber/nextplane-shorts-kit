## Query
```sql
with base as (
  select distinct on (upper(registration)) listing_id, upper(registration) reg, model, year, serial_number, asking_price_usd cur
  from aircraft_listings
  where status='active' and registration ~* '^N[0-9]' and location_country in ('US','USA') and category='piston_single'
    and last_seen_at > now() - interval '7 days' and asking_price_usd between 20000 and 3000000 and registration <> 'N12345'
    and upper(make) ~ 'BEECH' and upper(model) ~ 'G36' and year = 2008
  order by upper(registration), (source='controller') desc, last_seen_at desc)
select b.*, (select m.year_mfr||' '||m.serial_number from aircraft_master m where m.n_number = substr(b.reg,2) limit 1) faa
from base b order by cur;
```
