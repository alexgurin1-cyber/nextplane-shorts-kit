## Query
```sql
with base as materialized (
  select distinct on (upper(registration)) listing_id, upper(registration) reg, asking_price_usd cur
  from aircraft_listings
  where status='active' and registration ~* '^N[0-9]' and location_country in ('US','USA') and category='piston_single'
    and last_seen_at > now() - interval '7 days' and asking_price_usd between 20000 and 3000000 and registration <> 'N12345'
    and upper(make) ~ 'CESSNA' and upper(model) ~ '182'
  order by upper(registration), (source='controller') desc, last_seen_at desc
), h as (
  select b.*, (select ph.old_price from aircraft_price_history ph where ph.listing_id=b.listing_id order by recorded_at asc limit 1) first_ask
  from base b)
select count(*) n, count(*) filter (where first_ask>cur) cut_any,
  count(*) filter (where first_ask>cur and (first_ask-cur)::numeric/first_ask <= 0.35) cut_kept,
  percentile_cont(0.5) within group (order by first_ask-cur) filter (where first_ask>cur and (first_ask-cur)::numeric/first_ask <= 0.35) med_cut
from h;
```
